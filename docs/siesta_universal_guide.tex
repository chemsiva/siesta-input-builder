\section{Foundations of SIESTA \& First-Principles Physics}

\subsection{The Quantum Many-Body Challenge and Hohenberg-Kohn Theorems}
In condensed matter physics and quantum chemistry, the fundamental goal is to solve the time-independent Schr{\"o}dinger equation for a collection of $N$ interacting electrons and $M$ atomic nuclei:
\begin{equation}
\hat{H} \Psi(\mathbf{r}_1, \dots, \mathbf{r}_N; \mathbf{R}_1, \dots, \mathbf{R}_M) = E \Psi(\mathbf{r}_1, \dots, \mathbf{r}_N; \mathbf{R}_1, \dots, \mathbf{R}_M)
\end{equation}
Under the \textbf{Born-Oppenheimer approximation}, because atomic nuclei are thousands of times more massive than electrons ($M_{\text{nuc}} \gg m_e$), nuclear kinetic energy is decoupled, treating nuclei as static external point charges that produce a Coulomb potential $V_{\text{ext}}(\mathbf{r})$ felt by the electrons. 

Even with this simplification, the exact electronic wave function $\Psi(\mathbf{r}_1, \dots, \mathbf{r}_N)$ depends on $3N$ spatial coordinates. For a modest semiconductor slab containing 100 atoms ($\sim 1000$ valence electrons), $\Psi$ lives in a 3000-dimensional Hilbert space, rendering direct wave function approaches computationally impossible for solid-state materials.

\textbf{Density Functional Theory (DFT)} circumvents this exponential bottleneck through the two foundational theorems of \textbf{Hohenberg and Kohn (1964)}:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Theorem 1 (One-to-One Mapping):} The ground-state electron density $\rho(\mathbf{r})$ uniquely determines the external potential $V_{\text{ext}}(\mathbf{r})$ up to an additive constant, and therefore uniquely determines the full Hamiltonian and all ground-state observables of the many-body system.
    \item \textbf{Theorem 2 (Variational Principle):} There exists a universal energy functional $E[\rho]$ such that the exact ground-state density minimizes $E[\rho]$, yielding the true ground-state energy $E_0$:
    \begin{equation}
    E_0 \le E[\tilde{\rho}] \quad \text{for any trial density } \tilde{\rho}(\mathbf{r}) \ge 0, \quad \int \tilde{\rho}(\mathbf{r}) d^3\mathbf{r} = N
    \end{equation}
\end{enumerate}

\subsection{The Kohn-Sham Formulation}
To make this variational principle computationally tractable, \textbf{Kohn and Sham (1965)} replaced the intractable interacting electron system with an auxiliary, non-interacting electron system that yields the exact same ground-state density $\rho(\mathbf{r})$. The total energy functional is partitioned into four physically transparent terms:
\begin{equation}
E[\rho] = T_s[\rho] + \int \rho(\mathbf{r}) V_{\text{ext}}(\mathbf{r}) d^3\mathbf{r} + E_H[\rho] + E_{\text{xc}}[\rho]
\end{equation}
where:
\begin{itemize}[leftmargin=*]
    \item $T_s[\rho] = -\frac{\hbar^2}{2m} \sum_{i=1}^N \int \psi_i^*(\mathbf{r}) \nabla^2 \psi_i(\mathbf{r}) d^3\mathbf{r}$ is the kinetic energy of non-interacting electrons described by single-particle Kohn-Sham orbitals $\psi_i(\mathbf{r})$.
    \item $E_H[\rho] = \frac{e^2}{2} \iint \frac{\rho(\mathbf{r})\rho(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|} d^3\mathbf{r} d^3\mathbf{r}'$ is the classical electrostatic Hartree electron-electron repulsion.
    \item $E_{\text{xc}}[\rho]$ is the \textbf{Exchange-Correlation (XC) functional}, which encapsulates all quantum mechanical many-body effects: Pauli exchange repulsion and dynamic electron correlation.
\end{itemize}

Varying $E[\rho]$ with respect to $\psi_i^*(\mathbf{r})$ yields the canonical \textbf{Kohn-Sham Equations}:
\begin{equation}
\left( -\frac{\hbar^2}{2m}\nabla^2 + V_{\text{eff}}[\rho](\mathbf{r}) \right) \psi_i(\mathbf{r}) = \varepsilon_i \psi_i(\mathbf{r}), \quad \text{where } \rho(\mathbf{r}) = \sum_{i=1}^N |\psi_i(\mathbf{r})|^2
\end{equation}
with the effective single-particle potential defined as:
\begin{equation}
V_{\text{eff}}(\mathbf{r}) = V_{\text{ext}}(\mathbf{r}) + e^2 \int \frac{\rho(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|} d^3\mathbf{r}' + \frac{\delta E_{\text{xc}}[\rho]}{\delta \rho(\mathbf{r})}
\end{equation}
Because $V_{\text{eff}}$ depends on $\rho(\mathbf{r})$, which in turn depends on the unknown wavefunctions $\psi_i(\mathbf{r})$, the Kohn-Sham equations must be solved iteratively until self-consistency is achieved (\textbf{Self-Consistent Field, SCF}).

\subsection{Plane-Wave DFT vs. Linear Combination of Atomic Orbitals (LCAO)}
To solve the differential Kohn-Sham equations on a computer, the continuous orbitals $\psi_i(\mathbf{r})$ must be expanded in a discrete mathematical basis set:
\begin{equation}
\psi_i(\mathbf{r}) = \sum_{\mu} c_{\mu i} \phi_{\mu}(\mathbf{r})
\end{equation}
The choice of basis functions $\phi_{\mu}(\mathbf{r})$ fundamentally distinguishes computational DFT codes:

\begin{table}[h]
\centering
\small
\begin{tabular}{lll}
\toprule
\textbf{Feature} & \textbf{Plane-Wave Codes (QE, VASP)} & \textbf{LCAO Codes (SIESTA)} \\
\midrule
\textbf{Basis Function $\phi_{\mu}$} & Orthogonal Fourier waves: $e^{i(\mathbf{k}+\mathbf{G})\cdot\mathbf{r}}$ & Strictly localized numerical atomic orbitals: $R_{nl}(r) Y_{lm}(\hat{r})$ \\
\textbf{Systematic Convergence} & Trivial: increase energy cutoff $E_{\text{cut}}$ & Requires selecting radial cutoffs and polarization shells ($SZ, DZP, TZP$) \\
\textbf{Vacuum Efficiency} & Very poor: empty vacuum space requires plane waves everywhere & \textbf{Extremely efficient}: empty vacuum has zero orbitals $\implies$ zero compute cost \\
\textbf{Scaling with Size $N$} & Cubic $O(N^3)$ or $O(N^2 \log N)$ FFT cost & \textbf{Linear $O(N)$ or sparse $O(N)$ Hamiltonian} \\
\textbf{Sparsity} & Dense matrices in real space & \textbf{Strictly sparse Hamiltonian \& Overlap matrices} \\
\textbf{Ideal Applications} & Dense 3D metals, isotropic bulk solids & \textbf{Surface slabs, 2D layers, large clusters, nanoribbons, biomolecules} \\
\bottomrule
\end{tabular}
\caption{Comparison between plane-wave DFT methods and the localized atomic orbital approach in SIESTA.}
\end{table}

\subsection{Why SIESTA? The $O(N)$ Localized Orbital Paradigm}
SIESTA (\textit{Spanish Initiative for Electronic Simulations with Thousands of Atoms}) was conceived by Emilio Artacho, Alberto Garc{\'i}a, Julian Gale, Javier Junquera, Richard Martin, Pablo Ordej{\'o}n, Daniel S{\'a}nchez-Portal, and Jos{\'e} M. Soler. 

The defining breakthrough of SIESTA is that its basis orbitals $\phi_{\mu}(\mathbf{r})$ are \textbf{strictly zero} beyond a finite cutoff radius $r_{c,\mu}$:
\begin{equation}
\phi_{\mu}(\mathbf{r}) \equiv 0 \quad \text{for } |\mathbf{r} - \mathbf{R}_I| > r_{c,\mu}
\end{equation}
Because orbitals strictly vanish beyond $r_c$, the overlap matrix $S_{\mu\nu} = \langle \phi_{\mu} | \phi_{\nu} \rangle$ and Hamiltonian matrix $H_{\mu\nu} = \langle \phi_{\mu} | \hat{H} | \phi_{\nu} \rangle$ are \textbf{sparse banded matrices}: matrix elements are strictly zero unless two atoms are within distance $r_{c,\mu} + r_{c,\nu}$. 
As a direct consequence:
\begin{itemize}[leftmargin=*]
    \item Matrix construction scales strictly as $\mathcal{O}(N)$.
    \item Storing matrices requires memory proportional to $\mathcal{O}(N)$ rather than $\mathcal{O}(N^2)$.
    \item Systems comprising thousands of atoms (e.g. $FAPbI_3$ perovskite slabs with organic ligands) can be calculated on standard computing clusters where plane-wave codes would run out of memory.
\end{itemize}

\section{The FDF (Flexible Data Format) Language}

\subsection{Syntax, Tokens, and Data Blocks}
All control parameters, structural descriptions, and calculation directives in SIESTA are provided through a unified plain-text file known as the \textbf{FDF (Flexible Data Format)} file (conventionally named with the \texttt{.fdf} extension).

Key rules of the FDF format:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Case Insensitivity:} FDF keywords are case-insensitive. \texttt{MeshCutoff}, \texttt{meshcutoff}, and \texttt{Mesh.Cutoff} resolve to the exact same parameter.
    \item \textbf{Punctuation Tolerance:} Characters like periods (\texttt{.}), hyphens (\texttt{-}), and underscores (\texttt{\_}) in variable names are largely treated as interchangeable separators in modern SIESTA versions.
    \item \textbf{Single Values:} Specified as \texttt{Keyword Value [Unit]}:
\begin{lstlisting}
MeshCutoff         350.0 Ry
ElectronicTemperature  25.0 meV
\end{lstlisting}
    \item \textbf{Logical Values:} Booleans can be specified using \texttt{true}/\texttt{false}, \texttt{yes}/\texttt{no}, or single letters \texttt{T}/\texttt{F}.
    \item \textbf{Data Blocks:} Structured arrays, coordinate tables, and matrix specifications are enclosed in \texttt{\%block BlockName} and \texttt{\%endblock BlockName}:
\begin{lstlisting}
%block ChemicalSpeciesLabel
  1   14  Si      Si.psf
%endblock ChemicalSpeciesLabel
\end{lstlisting}
    \item \textbf{Comments:} Any text preceded by \texttt{\#} or \texttt{!} is treated as a comment and ignored by the parser.
\end{enumerate}

\subsection{Physical Dimensions and Unit Scaling}
SIESTA provides automatic physical unit conversion. A user can specify units in their preferred scientific convention, and SIESTA internally scales them to atomic units (Hartree, Bohr):

\section{Crystal Structure, Lattice \& Geometry Setup}

\subsection{Lattice Representation: Vectors vs. Parameters}
Every periodic calculation requires defining the unit cell Bravais lattice. In SIESTA, this is governed by the global scaling parameter \texttt{LatticeConstant} and one of two mutually exclusive blocks:

\subsubsection{Option A: Explicit Lattice Vectors (Recommended)}
\begin{lstlisting}
LatticeConstant 1.0 Ang
%block LatticeVectors
  12.640000   0.000000   0.000000
   0.000000  12.640000   0.000000
   0.000000   0.000000  36.000000
%endblock LatticeVectors
\end{lstlisting}
Here, each row represents the Cartesian components of basis vectors $\mathbf{a}_1, \mathbf{a}_2, \mathbf{a}_3$. Because \texttt{LatticeConstant} is set to \texttt{1.0 Ang}, the numbers inside the block represent absolute Cartesian lengths in \AA ngstr{\"o}ms.

\subsubsection{Option B: Crystallographic Lattice Parameters}
\begin{lstlisting}
LatticeConstant 5.4307 Ang
%block LatticeParameters
  1.0  1.0  1.0  90.0  90.0  90.0
%endblock LatticeParameters
\end{lstlisting}
The row defines $a/a_0, b/a_0, c/a_0, \alpha, \beta, \gamma$, where $\alpha$ is the angle between $\mathbf{a}_2$ and $\mathbf{a}_3$, $\beta$ between $\mathbf{a}_1$ and $\mathbf{a}_3$, and $\gamma$ between $\mathbf{a}_1$ and $\mathbf{a}_2$.

\subsection{Atomic Coordinates Formats}
The spatial positions of all atoms in the simulation cell are declared in the \texttt{\%block AtomicCoordinatesAndAtomicSpecies} block. The interpretation of these coordinates is dictated by \texttt{AtomicCoordinatesFormat}:
\begin{itemize}[leftmargin=*]
    \item \texttt{Fractional}: Dimensionless crystal coordinates $(u, v, w)$ such that position $\mathbf{R} = u \mathbf{a}_1 + v \mathbf{a}_2 + w \mathbf{a}_3$, with $0 \le u, v, w < 1$.
    \item \texttt{NotScaledCartesianAng}: Unscaled Cartesian coordinates $(x, y, z)$ in \AA ngstr{\"o}ms.
    \item \texttt{NotScaledCartesianBohr}: Unscaled Cartesian coordinates in atomic units (Bohr).
    \item \texttt{ScaledCartesian}: Cartesian coordinates multiplied by the global \texttt{LatticeConstant}.
\end{itemize}

\begin{lstlisting}
AtomicCoordinatesFormat Fractional
%block AtomicCoordinatesAndAtomicSpecies
  0.00000000  0.00000000  0.15000000  1   # Pb atom at bottom
  0.50000000  0.50000000  0.15000000  2   # I atom
%endblock AtomicCoordinatesAndAtomicSpecies
\end{lstlisting}
The fourth column is the integer species index corresponding to the sequential index in \texttt{\%block ChemicalSpeciesLabel}.

\subsection{Surface Slabs, Vacuum Padding, and Dipole Correction}
When modeling surfaces, interfaces, or 2D materials (graphene, perovskite nanosheets, transition metal dichalcogenides), three-dimensional periodic boundary conditions (PBC) are still enforced by the FFT grid. To prevent spurious electronic interactions between periodic vertical images of the slab along the surface normal ($z$), one must:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Add Sufficient Vacuum:} Typically $15\text{ \AA} \le d_{\text{vac}} \le 25\text{ \AA}$ between the highest atom of one slab and the lowest atom of the periodic replica.
    \item \textbf{Center the Slab:} Shift atomic coordinates so the vacuum is evenly split above and below the material (\texttt{AtomicCoordinatesOrigin COP-Z}).
    \item \textbf{Apply Dipole Correction:} If the slab is asymmetric (e.g., polar surface termination or molecule adsorbed on one side only), an artificial net dipole moment $\mathbf{p}$ arises across the cell. This induces a fictitious linear electric field across the vacuum. Setting:
\begin{lstlisting}
Slab.DipoleCorrection true
\end{lstlisting}
    instructs SIESTA to introduce an internal compensating dipole sheet in the vacuum gap, flattening the potential and yielding exact, uncorrupted work functions.
\end{enumerate}

\section{Pseudopotentials \& PAO Basis Sets}

\subsection{The Pseudopotential Concept (Norm-Conservation \& Kleinman-Bylander)}
Core electrons (e.g. $1s^2 2s^2 2p^6$ in Silicon) are tightly bound to the nucleus and chemically inert. However, their wavefunctions oscillate violently near the core to maintain orthogonality with deeper shells. In pseudopotential theory:
\begin{enumerate}[leftmargin=*]
    \item Core electrons and the nuclear Coulomb potential are replaced by an effective smooth \textbf{pseudopotential} $V_{\text{pseudo}}(r)$.
    \item Beyond a core radius $r_c$, the pseudo-wavefunction $\tilde{\psi}(r)$ matches the all-electron wavefunction $\psi(r)$ exactly.
    \item Under \textbf{norm-conservation}, the integral of $|\tilde{\psi}(r)|^2$ inside $r_c$ equals that of the all-electron wavefunction, ensuring correct electrostatic scattering properties across varying chemical environments.
\end{enumerate}
SIESTA requires norm-conserving pseudopotentials in the separable \textbf{Kleinman-Bylander (KB)} form:
\begin{equation}
\hat{V}_{\text{NL}} = \sum_{l,m} \frac{| \chi_{lm} \rangle \langle \chi_{lm} |}{\langle \chi_{lm} | \delta V_l | \chi_{lm} \rangle}
\end{equation}

\subsection{Pseudopotential File Formats: PSF and Modern PSML}
\begin{itemize}[leftmargin=*]
    \item \textbf{PSF (Traditional ASCII Format):} Stores radial grid data, logarithmic derivatives, and local/non-local projectors. Named \texttt{[Element].psf} (e.g., \texttt{Pb.psf}, \texttt{I.psf}).
    \item \textbf{PSML (Pseudopotential Markup Language):} The modern XML-based standard adopted by SIESTA 5.x. Enables full provenance tracking, relativistic spin-orbit projectors, and semicore configuration verification.
\end{itemize}

\subsection{Numerical Atomic Orbitals (NAOs) and Basis Cardinality}
Unlike Gaussian orbitals used in quantum chemistry, SIESTA computes radial basis functions numerically by solving the Schr{\"o}dinger equation for the isolated pseudo-atom enclosed in a spherical confinement well:
\begin{equation}
\phi_{nlm}(\mathbf{r}) = R_{nl}(r) Y_{lm}(\hat{\mathbf{r}})
\end{equation}
The basis set hierarchy is defined by \texttt{PAO.BasisSize}:
\begin{itemize}[leftmargin=*]
    \item \textbf{\texttt{SZ} (Single-Zeta):} Minimal basis. Exactly one radial function per valence orbital (e.g. for Carbon $2s, 2p$: 1 $s$ function and 3 $p$ functions = 4 basis functions). Fast, qualitative.
    \item \textbf{\texttt{DZ} (Double-Zeta):} Two radial functions per valence orbital ($2s, 2s', 2p, 2p'$). Essential for chemical bonding and valence deformation.
    \item \textbf{\texttt{DZP} (Double-Zeta Polarized):} \textbf{The standard for production research.} Takes \texttt{DZ} and adds a polarization shell with angular momentum $l+1$ (e.g. adds $3d$ orbitals to Carbon $2p$). Allows electron clouds to distort under asymmetric surface fields.
    \item \textbf{\texttt{TZP} / \texttt{TZ2P}:} Triple-zeta with multiple polarization shells. High-precision limit, approaching the complete basis set (CBS) plane-wave accuracy.
\end{itemize}

\subsection{The Spatial Confinement Parameter: \texttt{PAO.EnergyShift}}
How does SIESTA determine where to truncate the orbital so it strictly equals zero at $r_c$?
Instead of forcing the user to guess radii for every shell of every element, SIESTA provides the universal thermodynamic parameter \texttt{PAO.EnergyShift}:
\begin{equation}
\delta E = \frac{\hbar^2 \pi^2}{2 m_e r_{c,nl}^2}
\end{equation}
\begin{itemize}[leftmargin=*]
    \item \textbf{Physical Meaning:} It represents the energy increase experienced by an electron when it is confined inside a spherical box of radius $r_c$.
    \item \textbf{Standard Value:} Default is \texttt{100 meV} to \texttt{250 meV}.
    \item \textbf{Behavior:} A \textit{smaller} energy shift (e.g. \texttt{50 meV}) gives a larger $r_c$, expanding the basis set to capture diffuse vacuum states at the cost of denser matrices. A \textit{larger} energy shift (e.g. \texttt{300 meV}) makes orbitals more compact and faster, but may under-represent surface dipole relaxation.
\end{itemize}

\section{Real-Space Grid \& Exchange-Correlation (XC)}

\subsection{The Real-Space Mesh and \texttt{MeshCutoff}}
While Hamiltonian and overlap matrices are computed on atomic basis orbitals, the electron density $\rho(\mathbf{r})$ and Hartree/XC potentials are represented on a 3D regular Cartesian real-space grid.
The grid fineness is specified by \texttt{MeshCutoff}:
\begin{equation}
E_{\text{cut}} = \frac{\hbar^2 k_{\text{max}}^2}{2 m_e} \implies \Delta x \approx \frac{\pi}{k_{\text{max}}} = \frac{\pi}{\sqrt{2 m_e E_{\text{cut}}}}
\end{equation}
\begin{itemize}[leftmargin=*]
    \item For $E_{\text{cut}} = 200\text{ Ry} \implies \Delta x \approx 0.12\text{ \AA}$.
    \item For $E_{\text{cut}} = 350\text{ Ry} \implies \Delta x \approx 0.09\text{ \AA}$.
    \item \textbf{Rule of Thumb:} Always perform a mesh cutoff convergence test. Standard semiconductors require $250\text{--}350\text{ Ry}$. Highly localized elements ($O, F, Fe$) require $350\text{--}450\text{ Ry}$.
\end{itemize}

\subsection{The Eggbox Effect}
Because the real-space grid is discrete and does not automatically displace with the atoms as they move, shifting an atom across grid points induces an artificial periodic ripple in the total energy and forces, termed the \textbf{eggbox effect}:
\begin{itemize}[leftmargin=*]
    \item \textbf{Remedy 1:} Increase \texttt{MeshCutoff} ($> 300\text{ Ry}$).
    \item \textbf{Remedy 2:} Use modern grid-cell sampling (\texttt{GridCellSampling true}).
    \item \textbf{Remedy 3:} Use soft confinement potentials (\texttt{PAO.SoftDefault true}) to smooth the orbital second derivatives near $r_c$.
\end{itemize}

\subsection{Exchange-Correlation (XC) Functionals}
SIESTA supports both local and gradient-corrected approximations:
\begin{lstlisting}
XC.functional   GGA
XC.authors      PBE
\end{lstlisting}
\begin{itemize}[leftmargin=*]
    \item \textbf{LDA (Local Density Approximation):} \texttt{XC.functional LDA}, \texttt{XC.authors CA} (Ceperley-Alder parameterized by Perdew-Zunger). Reliable for bulk moduli, underestimates lattice parameters by $1\text{--}2\%$.
    \item \textbf{GGA (Generalized Gradient Approximation):} \texttt{XC.functional GGA}, \texttt{XC.authors PBE} (Perdew-Burke-Ernzerhof). The workhorse functional for solid-state chemistry and surfaces.
    \item \textbf{PBEsol:} \texttt{XC.authors PBEsol}. Refined GGA specifically tuned for equilibrium lattice constants and bulk phonon frequencies.
    \item \textbf{van der Waals Dispersion (Grimme DFT-D3):} Crucial for molecular adsorption, perovskite organic-inorganic interfaces, and layered materials:
\begin{lstlisting}

\end{lstlisting}
\end{itemize}

\section{Self-Consistent Field (SCF) Control}

\subsection{The Density Matrix Formulation}
Instead of dealing with individual Kohn-Sham wavefunctions, SIESTA expresses all physical observables via the single-particle \textbf{density matrix} $P_{\mu\nu}$:
\begin{equation}
P_{\mu\nu} = \sum_i f_i c_{\mu i} c_{\nu i}^*
\end{equation}
where $f_i$ is the Fermi-Dirac occupation factor. The real-space electron density is reconstructed as:
\begin{equation}
\rho(\mathbf{r}) = \sum_{\mu,\nu} P_{\mu\nu} \phi_{\mu}^*(\mathbf{r}) \phi_{\nu}(\mathbf{r})
\end{equation}

\subsection{SCF Convergence Criteria and Mixing Algorithms}
During each SCF iteration $m$, the computed density matrix $P^{(m)}$ differs from the input density matrix $P_{\text{in}}^{(m)}$. A simple substitution $P_{\text{in}}^{(m+1)} = P^{(m)}$ leads to violent charge sloshing and divergence. Therefore, mixing algorithms are applied:

\begin{lstlisting}
MaxSCFIterations       150
DM.Tolerance           1.0d-4
DM.MixingWeight        0.25
DM.NumberPulay         8
\end{lstlisting}
\begin{itemize}[leftmargin=*]
    \item \textbf{\texttt{DM.Tolerance}:} The calculation is declared converged when:
    \begin{equation}
    \max_{\mu,\nu} |P_{\mu\nu}^{(m)} - P_{\mu\nu}^{(m-1)}| < \texttt{DM.Tolerance} \quad (\text{typically } 10^{-4}\text{ to } 10^{-5})
    \end{equation}
    \item \textbf{\texttt{DM.MixingWeight} ($\alpha$):} Fraction of the newly calculated density matrix mixed with the old matrix. For difficult systems (surfaces, transition metals, low bandgap perovskites), lower this value to \texttt{0.05} or \texttt{0.10} to prevent divergence.
    \item \textbf{\texttt{DM.NumberPulay}:} Depth of the Pulay Direct Inversion in the Iterative Subspace (DIIS) history. Using 6 to 8 history steps accelerates convergence exponentially after the initial linear regime.
\end{itemize}

\subsection{Electronic Temperature (Fermi Smearing)}
For metallic systems or surfaces with near-zero bandgaps, sharp discontinuities in Fermi occupations at $T=0$ destabilize SCF convergence. Introducing a finite electronic temperature smears occupations via the Fermi-Dirac function:
\begin{equation}
f(\varepsilon) = \frac{1}{e^{(\varepsilon - E_F)/k_B T_{\text{elec}}} + 1}
\end{equation}
\begin{lstlisting}
ElectronicTemperature  25.0 meV
\end{lstlisting}
This represents $T \approx 290\text{ K}$, smoothing occupation transitions and ensuring rapid, monotonic convergence.

\section{Brillouin Zone Sampling \& K-Points}

\subsection{Reciprocal Space Integration \& Monkhorst-Pack Grids}
In crystalline solids, translational symmetry dictates that wavefunctions obey Bloch's theorem: $\psi_{n\mathbf{k}}(\mathbf{r}) = e^{i\mathbf{k}\cdot\mathbf{r}} u_{n\mathbf{k}}(\mathbf{r})$. Physical properties require integration over the first Brillouin zone:
\begin{equation}
\bar{g} = \frac{\Omega}{(2\pi)^3} \int_{\text{BZ}} g(\mathbf{k}) d^3\mathbf{k} \approx \sum_{\mathbf{k}} w_{\mathbf{k}} g(\mathbf{k})
\end{equation}
SIESTA generates discrete $k$-point grids using the \textbf{Monkhorst-Pack (MP)} scheme:
\begin{lstlisting}
%block kgrid_Monkhorst_Pack
  4   0   0   0.5
  0   4   0   0.5
  0   0   1   0.0
%endblock kgrid_Monkhorst_Pack
\end{lstlisting}
The first three numbers in each row define the divisions along reciprocal vectors $\mathbf{b}_1, \mathbf{b}_2, \mathbf{b}_3$. The fourth number is the origin shift (typically \texttt{0.5} to avoid high-symmetry points or \texttt{0.0} for $\Gamma$-centered meshes).

\subsection{Dimensionality Rules for K-Points}
\begin{itemize}[leftmargin=*]
    \item \textbf{3D Bulk Crystals (e.g. Silicon, Perovskite bulk):} Full 3D grid ($6 \times 6 \times 6$ or $8 \times 8 \times 8$).
    \item \textbf{2D Surfaces \& Slabs:} \textbf{Strictly 1 point along $z$} ($k_z = 1$, e.g. $4 \times 4 \times 1$). Because the slab has vacuum along $z$, there is zero physical dispersion along the reciprocal $z$-axis. Setting $k_z > 1$ wastes massive compute power with zero physical benefit.
    \item \textbf{1D Nanowires / Nanotubes:} 1 point along $x$ and $y$ ($1 \times 1 \times 8$).
    \item \textbf{0D Molecules / Isolated Clusters:} Single $\Gamma$-point calculation ($1 \times 1 \times 1$).
\end{itemize}

\section{Structural Relaxation, General Constraints \& Molecular Dynamics}

\subsection{Geometry Optimization Algorithms}
Structural relaxation minimizes the total energy with respect to nuclear positions and cell vectors until atomic forces and stress tensors fall below strict tolerances:
\begin{lstlisting}
MD.TypeOfRun       CG
MD.NumCGsteps      200
MD.MaxForceTol     0.04 eV/Ang
MD.MaxStressTol    1.0 GPa
MD.VariableCell    false
\end{lstlisting}
\begin{itemize}[leftmargin=*]
    \item \textbf{\texttt{MD.TypeOfRun}:} Options include \texttt{CG} (Conjugate Gradient, standard), \texttt{Broyden} (quasi-Newton), and \texttt{FIRE} (Fast Inertial Relaxation Engine, superior for complex interfaces and loose initial configurations).
    \item \textbf{\texttt{MD.MaxForceTol}:} The maximum force component allowed on any atom for convergence (typically $0.02\text{--}0.04\text{ eV/\AA}$).
    \item \textbf{\texttt{MD.VariableCell}:} Set to \texttt{true} to relax cell vectors simultaneously. \textbf{Critical Warning:} Never enable \texttt{MD.VariableCell true} on a surface slab without fixing the $c$-axis, or the simulation will compress the vacuum gap!
\end{itemize}

\subsection{SIESTA Section 7.7: Use of General Constraints}
SIESTA provides a versatile constraints engine (\texttt{\%block Geometry.Constraints}) to freeze or couple degrees of freedom during relaxation:

\begin{lstlisting}
%block Geometry.Constraints
  atom [1 -- 12]              # Freeze atoms 1 through 12 completely
  atom 15  0.  0.  1.         # Fix atom 15 along Z only (allow XY relaxation)
  cell-vector C               # Freeze out-of-plane vacuum axis
  stress XX YY                # Constrain in-plane stress components
  Z 82                        # Freeze all Lead (Pb) atoms by atomic number
%endblock Geometry.Constraints
\end{lstlisting}

\subsubsection{Key Constraint Syntax Rules}
\begin{itemize}[leftmargin=*]
    \item \textbf{Range specifications:} Supports \texttt{atom [1 -- 4]}, \texttt{atom from 1 to 4 step 1}, and space-separated lists \texttt{atom 1 2 3 4}.
    \item \textbf{Directional Constraints:} Append three real numbers representing a Cartesian projection vector. \textbf{Crucial Rule:} Reals \textit{must} include a trailing decimal point (e.g., \texttt{0. 0. 1.}). SIESTA subtracts the force component projected onto this vector:
    \begin{equation}
    \mathbf{F}_{\text{active}} = \mathbf{F} - (\mathbf{F} \cdot \hat{\mathbf{n}}) \hat{\mathbf{n}}
    \end{equation}
    \item \textbf{Cell Constraints:} \texttt{cell-vector C} locks the vacuum height, allowing in-plane cell vectors $A$ and $B$ to relax without collapsing the vacuum.
\end{itemize}

\subsection{Projected Density of States (PDOS)}
Orbital-resolved density of states $g_{\mu}(E) = \sum_{n,\mathbf{k}} w_{\mathbf{k}} |\langle \phi_{\mu} | \psi_{n\mathbf{k}} \rangle|^2 \delta(E - E_{n\mathbf{k}})$:

%endblock ProjectedDensityOfStates

\subsection{Work Function Determination ($\Phi = V_{\text{vac}} - E_F$)}
The work function $\Phi$ is the minimum energy required to remove an electron from the Fermi level of a solid to the electrostatic vacuum plateau outside the surface:
\begin{equation}
\Phi = V_{\text{vac}} - E_F
\end{equation}
\begin{enumerate}[leftmargin=*]
    \item Request the electrostatic potential grid: \texttt{SaveElectrostaticPotential true} $\implies$ generates \texttt{SystemLabel.VH}.
    \item Compute planar average $\bar{V}(z) = \frac{1}{A} \iint V(\mathbf{r}) dx dy$ using \texttt{Util/Macroave} or \texttt{grid2cube}.
    \item Identify the asymptotic flat plateau $V_{\text{vac}}$ in the middle of the vacuum region.
    \item Subtract the self-consistent Fermi level: $\Phi = V_{\text{vac}} - E_F$.
\end{enumerate}

\subsection{Tutorial 3: Gas Adsorption on Surface — Energy \& Charge Density Difference}
\textbf{Objective:} Calculate the binding energy $E_{\text{ads}}$ of a water molecule ($H_2O$) on a $TiO_2$ anatase surface, and compute the planar charge transfer $\Delta\rho(z)$.

\subsubsection{The Three Independent Calculations}
To compute the adsorption energy:
\begin{equation}
E_{\text{ads}} = E_{\text{total}}[\text{Slab}+\text{Mol}] - E_{\text{slab}}[\text{Clean Slab}] - E_{\text{mol}}[\text{Isolated Mol}]
\end{equation}
\begin{enumerate}[leftmargin=*]
    \item \textbf{System 1 (Total):} $H_2O$ adsorbed on $TiO_2$ slab in cell $a \times b \times c$. Output: $E_{\text{total}} = -14256.789\text{ eV}$.
    \item \textbf{System 2 (Clean Slab):} Bare $TiO_2$ slab in the exact same cell $a \times b \times c$. Output: $E_{\text{slab}} = -13850.123\text{ eV}$.
    \item \textbf{System 3 (Free Molecule):} Isolated $H_2O$ molecule in the exact same cell. Output: $E_{\text{mol}} = -405.892\text{ eV}$.
\end{enumerate}

\begin{equation}
E_{\text{ads}} = -14256.789 - (-13850.123) - (-405.892) = -0.774\text{ eV} \quad (\text{Exothermic Physisorption})
\end{equation}

\subsubsection{Planar Average Charge Density Difference (CDD)}
Request valence charge densities: \texttt{SaveRho true}. Generate 3D grid files:
\begin{equation}
\Delta\rho(\mathbf{r}) = \rho_{\text{total}}(\mathbf{r}) - \rho_{\text{slab}}(\mathbf{r}) - \rho_{\text{mol}}(\mathbf{r})
\end{equation}
Integrate in the $xy$-plane to yield the planar profile:
\begin{equation}
\Delta\bar{\rho}(z) = \frac{1}{A} \iint \Delta\rho(x,y,z) dx dy
\end{equation}
and the cumulative charge transfer across the interface:
\begin{equation}
\Delta Q(z) = \int_{-\infty}^z \Delta\bar{\rho}(z') dz'
\end{equation}

In **\textcircled{12} Surface \& Work Function Studio $\to$ Planar Average CDD**, uploading the resulting \texttt{.cube} or \texttt{.dat} file renders the dual-axis chart:
\begin{itemize}[leftmargin=*]
    \item Peak accumulation region ($+\Delta\rho$: electron buildup at the interface bond).
    \item Peak depletion region ($-\Delta\rho$: charge loss from the substrate surface).
    \item Net transferred charge $\Delta q$ and interface dipole $\Delta\mu$ in Debye.
\end{itemize}
